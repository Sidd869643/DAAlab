#include <stdio.h>

void merge(int a[], int low, int mid, int high) {
    int temp[100], i = low, j = mid + 1, k = 0;

    while (i <= mid && j <= high) {
        if (a[i] <= a[j]) temp[k++] = a[i++];
        else temp[k++] = a[j++];
    }
    while (i <= mid) temp[k++] = a[i++];
    while (j <= high) temp[k++] = a[j++];

    for (i = low, k = 0; i <= high; i++, k++)
        a[i] = temp[k];
}

void mergeSort(int a[], int low, int high) {
    int mid;
    if (low < high) {
        mid = (low + high) / 2;
        mergeSort(a, low, mid);
        mergeSort(a, mid + 1, high);
        merge(a, low, mid, high);
    }
}

int main() {
    int a[100], n, i;

    scanf("%d", &n);
    printf("Enter elements: ");
    for (i = 0; i < n; i++) scanf("%d", &a[i]);

5. Quick Sort

File: 04_quick_sort.c

#include <stdio.h>

int partition(int a[], int low, int high) {
    int pivot = a[high], i = low - 1, j, temp;

    for (j = low; j < high; j++) {
        if (a[j] <= pivot) {
            i++;
            temp = a[i]; a[i] = a[j]; a[j] = temp;
        }
    }
    temp = a[i + 1]; a[i + 1] = a[high]; a[high] = temp;
    return i + 1;
}

void quickSort(int a[], int low, int high) {
    int p;
    if (low < high) {
        p = partition(a, low, high);
        quickSort(a, low, p - 1);
        quickSort(a, p + 1, high);
    }
}

int main() {
    int a[100], n, i;
    printf("Enter number of elements: ");
    scanf("%d", &n);
    printf("Enter elements: ");
    for (i = 0; i < n; i++) scanf("%d", &a[i]);

    quickSort(a, 0, n - 1);

    printf("Sorted array: ");
    for (i = 0; i < n; i++) printf("%d ", a[i]);

#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int disk[100];
    int top;
    char name;
} Peg;

void moveDisk(Peg *a, Peg *b) {
    int diskA = (a->top == 0) ? 100000 : a->disk[a->top];
    int diskB = (b->top == 0) ? 100000 : b->disk[b->top];

    if (diskA < diskB) {
        b->disk[++b->top] = a->disk[a->top--];
        printf("Move disk %d from %c to %c\n", b->disk[b->top], a->name, b->name);
    } else {
        a->disk[++a->top] = b->disk[b->top--];
        printf("Move disk %d from %c to %c\n", a->disk[a->top], b->name, a->name);
    }
}

int main() {
    Peg s = {{0}, 0, 'A'}, a = {{0}, 0, 'B'}, d = {{0}, 0, 'C'};
    Peg *src = &s, *aux = &a, *dst = &d, *temp;
    int n, i, total;

    printf("Enter number of disks: ");
    scanf("%d", &n);
    if (n <= 0 || n >= 100) return 0;

    for (i = n; i >= 1; i--) s.disk[++s.top] = i;
    total = (1 << n) - 1;

    if (n % 2 == 0) {
        temp = dst; dst = aux; aux = temp;
    }

    for (i = 1; i <= total; i++) {
        if (i % 3 == 1) moveDisk(src, dst);
        else if (i % 3 == 2) moveDisk(src, aux);
        else moveDisk(aux, dst);
    }
    return 0;
}

9. Maximum Subarray

File: 08_maximum_subarray.c

#include <stdio.h>

int max(int a, int b) { return a > b ? a : b; }

int maxCrossing(int a[], int low, int mid, int high) {
    int sum = 0, left = -1000000000, right = -1000000000, i;

    for (i = mid; i >= low; i--) {
        sum += a[i];
        if (sum > left) left = sum;
    }
    sum = 0;
    for (i = mid + 1; i <= high; i++) {
        sum += a[i];
        if (sum > right) right = sum;
    }
    return left + right;
}

int maxSubarray(int a[], int low, int high) {
    int mid, x, y, z;
    if (low == high) return a[low];

    mid = (low + high) / 2;
    x = maxSubarray(a, low, mid);
    y = maxSubarray(a, mid + 1, high);
    z = maxCrossing(a, low, mid, high);
    return max(max(x, y), z);
}

int main() {
    int a[100], n, i;
    printf("Enter number of elements: ");
    scanf("%d", &n);
    printf("Enter array (may contain negative values): ");
    for (i = 0; i < n; i++) scanf("%d", &a[i]);

    printf("Maximum subarray sum = %d\n", maxSubarray(a, 0, n - 1));
    return 0;
}

10. Fractional Knapsack

File: 09_fractional_knapsack.c

#include <stdio.h>
#include <stdlib.h>

typedef struct {
    float weight, profit, ratio;
} Item;

int compare(const void *a, const void *b) {
    float x = ((Item *)a)->ratio;
    float y = ((Item *)b)->ratio;
    return (x < y) - (x > y); /* descending ratio */
}

int main() {
    Item item[100], temp;
    int n, i;
    float capacity, total = 0, take;

    printf("Enter number of items: ");
    scanf("%d", &n);
    for (i = 0; i < n; i++) {
        printf("Enter weight and profit of item %d: ", i + 1);
        scanf("%f%f", &item[i].weight, &item[i].profit);
        item[i].ratio = item[i].profit / item[i].weight;
    }
    printf("Enter knapsack capacity: ");
    scanf("%f", &capacity);

    qsort(item, n, sizeof(Item), compare);

    for (i = 0; i < n && capacity > 0; i++) {
        take = item[i].weight < capacity ? item[i].weight : capacity;
        total += take * item[i].ratio;
        capacity -= take;
    }

    printf("Maximum profit = %.2f\n", total);
    return 0;
}

12. Huffman Coding

File: 11_huffman_coding.c

#include <stdio.h>

#define MAX 100

typedef struct {
    int weight, parent, left, right;
} Node;

void printCode(Node tree[], int root, char code[], int top) {
    if (tree[root].left == -1 && tree[root].right == -1) {
        int i;
        printf("Symbol %d: ", root + 1);
        for (i = 0; i < top; i++) printf("%c", code[i]);
        printf("\n");
        return;
    }
    if (tree[root].left != -1) {
        code[top] = '0';
        printCode(tree, tree[root].left, code, top + 1);
    }
    if (tree[root].right != -1) {
        code[top] = '1';
        printCode(tree, tree[root].right, code, top + 1);
    }
}

int main() {
    Node t[2 * MAX];
    int n, i, j, x, y, m, min1, min2;
    char code[MAX];

    printf("Enter number of symbols: ");
    scanf("%d", &n);
    if (n < 2 || n > MAX) return 0;

    for (i = 0; i < 2 * n - 1; i++) {
        t[i].parent = t[i].left = t[i].right = -1;
        t[i].weight = 0;
    }
    for (i = 0; i < n; i++) {
        printf("Enter frequency of symbol %d: ", i + 1);
        scanf("%d", &t[i].weight);
    }

    for (m = n; m < 2 * n - 1; m++) {
        min1 = min2 = 1000000000;
        x = y = -1;
        for (i = 0; i < m; i++) {
            if (t[i].parent == -1 && t[i].weight < min1) {
                min2 = min1; y = x;
                min1 = t[i].weight; x = i;
            } else if (t[i].parent == -1 && t[i].weight < min2) {
                min2 = t[i].weight; y = i;
            }
        }
        t[x].parent = t[y].parent = m;
        t[m].weight = t[x].weight + t[y].weight;
        t[m].left = x;
        t[m].right = y;
    }

    printf("Huffman codes:\n");
    printCode(t, 2 * n - 2, code, 0);
    return 0;
}

int main() {
    int wt[100], val[100], dp[101][1001];
    int n, W, i, w;

    printf("Enter number of items: ");
    scanf("%d", &n);
    for (i = 1; i <= n; i++) {
        printf("Enter weight and value of item %d: ", i);

        for (w = 0; w <= W; w++) {
            if (i == 0 || w == 0) dp[i][w] = 0;
            else if (wt[i] <= w)
                dp[i][w] = max(val[i] + dp[i - 1][w - wt[i]], dp[i - 1][w]);

                dp[i][w] = dp[i - 1][w];
        }

    return 0;
}

15. Topological Sort

#include <stdio.h>

int main() {
    int g[20][20] = {0}, indegree[20] = {0}, q[20];
    int n, e, u, v, i, front = 0, rear = 0, count = 0;

    printf("Enter vertices and directed edges: ");
    scanf("%d%d", &n, &e);
    for (i = 0; i < e; i++) {
        scanf("%d%d", &u, &v);
        g[u][v] = 1;
        indegree[v]++;
    }

    for (i = 0; i < n; i++)
        if (indegree[i] == 0) q[rear++] = i;

    printf("Topological order: ");
    while (front < rear) {
        u = q[front++];
        printf("%d ", u);
        count++;
        for (v = 0; v < n; v++)
            if (g[u][v] && --indegree[v] == 0)
                q[rear++] = v;
    }

    if (count != n) printf("\nGraph contains a cycle; topological sort is impossible.");
    return 0;
}

#include <stdio.h>

int g[20][20], color[20], n;

int dfs(int u) {
    int v;
    color[u] = 1; /* currently in recursion stack */

    for (v = 0; v < n; v++) {
        if (g[u][v]) {
            if (color[v] == 1) return 1;
            if (color[v] == 0 && dfs(v)) return 1;
        }
    }
    color[u] = 2; /* completely processed */
    return 0;
}

            break;

17. Biconnected Components

File: 16_biconnected_components.c

#include <stdio.h>

#define N 20
int g[N][N], disc[N], low[N], parent[N], timer = 0;
int stackU[400], stackV[400], top = -1, n;

void push(int u, int v) {
    stackU[++top] = u;
    stackV[top] = v;
}

void printComponent(int u, int v) {
    printf("{ ");
    while (top >= 0) {
        int a = stackU[top], b = stackV[top--];
        printf("(%d-%d) ", a, b);
        if (a == u && b == v) break;
    }
    printf("}\n");
}

void dfs(int u) {
    int v;
    disc[u] = low[u] = ++timer;

    for (v = 0; v < n; v++) {
        if (!g[u][v]) continue;

        if (disc[v] == 0) {
            parent[v] = u;
            push(u, v);
            dfs(v);
            if (low[v] < low[u]) low[u] = low[v];

            if (low[v] >= disc[u])
                printComponent(u, v);
        } else if (v != parent[u] && disc[v] < disc[u]) {
            if (disc[v] < low[u]) low[u] = disc[v];
            push(u, v);
        }
    }
}

int main() {
    int e, u, v, i;
    printf("Enter vertices and undirected edges: ");
    scanf("%d%d", &n, &e);
    for (i = 0; i < e; i++) {
        scanf("%d%d", &u, &v);
        g[u][v] = g[v][u] = 1;
    }

    printf("Biconnected components (as edge sets):\n");
    for (i = 0; i < n; i++) {
        if (disc[i] == 0) {
            dfs(i);
            if (top >= 0) printComponent(stackU[0], stackV[0]);
        }
    }
    return 0;
}

18. Bellman Ford

#include <stdio.h>

#define INF 1000000000

typedef struct {
    int u, v, w;
} Edge;

int main() {
    Edge edge[200];
    int dist[50], n, e, src, i, j, changed;

    printf("Enter vertices and edges: ");
    scanf("%d%d", &n, &e);
    for (i = 0; i < e; i++) {
        printf("Enter source destination weight: ");
        scanf("%d%d%d", &edge[i].u, &edge[i].v, &edge[i].w);
    }
    printf("Enter source vertex: ");
    scanf("%d", &src);

    for (i = 0; i < n; i++) dist[i] = INF;
    dist[src] = 0;

    for (i = 1; i < n; i++) {
        changed = 0;
        for (j = 0; j < e; j++)
            if (dist[edge[j].u] != INF &&
                dist[edge[j].u] + edge[j].w < dist[edge[j].v]) {
                dist[edge[j].v] = dist[edge[j].u] + edge[j].w;
                changed = 1;
            }
        if (!changed) break;
    }

    for (j = 0; j < e; j++)
        if (dist[edge[j].u] != INF &&
            dist[edge[j].u] + edge[j].w < dist[edge[j].v]) {
            printf("Negative weight cycle reachable from source.\n");
            return 0;
        }

    printf("Shortest distances from %d:\n", src);
    for (i = 0; i < n; i++)
        if (dist[i] == INF) printf("%d: INF\n", i);
        else printf("%d: %d\n", i, dist[i]);
    return 0;
}

19. Ford Fulkerson

File: 18_ford_fulkerson.c

#include <stdio.h>

#define N 20
int residual[N][N], parent[N], n;

int bfs(int source, int sink) {
    int q[N], front = 0, rear = 0, visited[N] = {0}, u, v;
    q[rear++] = source;
    visited[source] = 1;
    parent[source] = -1;

    while (front < rear) {
        u = q[front++];
        for (v = 0; v < n; v++)
            if (!visited[v] && residual[u][v] > 0) {
                q[rear++] = v;
                parent[v] = u;
                visited[v] = 1;
            }
    }
    return visited[sink];
}

int main() {
    int e, u, v, capacity, source, sink, maxFlow = 0, pathFlow, cur, prev;

    printf("Enter vertices and edges: ");
    scanf("%d%d", &n, &e);
    for (int i = 0; i < e; i++) {
        printf("Enter from to capacity: ");
        scanf("%d%d%d", &u, &v, &capacity);
        residual[u][v] += capacity;
    }
    printf("Enter source and sink: ");
    scanf("%d%d", &source, &sink);

    while (bfs(source, sink)) {
        pathFlow = 1000000000;
        for (cur = sink; cur != source; cur = parent[cur]) {
            prev = parent[cur];
            if (residual[prev][cur] < pathFlow)
                pathFlow = residual[prev][cur];
        }

        for (cur = sink; cur != source; cur = parent[cur]) {
            prev = parent[cur];
            residual[prev][cur] -= pathFlow;
            residual[cur][prev] += pathFlow;
        }
        maxFlow += pathFlow;
    }

    printf("Maximum flow = %d\n", maxFlow);
    return 0;
}

20. N Queens Branch Bound

File: 19_n_queens_branch_bound.c

#include <stdio.h>

#define N 20
int n, board[N], col[N], diag1[2 * N], diag2[2 * N], found = 0;

void solve(int row) {
    int c, i;
    if (row == n) {
        found = 1;
        for (i = 0; i < n; i++) {
            for (c = 0; c < n; c++)
                printf("%c ", board[i] == c ? 'Q' : '.');
            printf("\n");
        }
        return;
    }

    for (c = 0; c < n; c++) {
        if (!col[c] && !diag1[row - c + n] && !diag2[row + c]) {
            board[row] = c;
            col[c] = diag1[row - c + n] = diag2[row + c] = 1;
            solve(row + 1);
            col[c] = diag1[row - c + n] = diag2[row + c] = 0;
        }
    }
}

#include <stdio.h>
#include <stdlib.h>

#define N 4
#define FOUND -1
int goal[16], path[100], solutionDepth = 0, nodes = 0;

int manhattan(int a[]) {
    int i, value, r, c, gr, gc, sum = 0;
    for (i = 0; i < 16; i++) {
        value = a[i];
        if (value == 0) continue;
        r = i / 4; c = i % 4;
        gr = (value - 1) / 4; gc = (value - 1) % 4;
        sum += abs(r - gr) + abs(c - gc);
    }
    return sum;
}

int dfs(int a[], int blank, int g, int bound, int previousMove) {
    int h = manhattan(a), f = g + h;
    int nextBound = 1000000, r = blank / 4, c = blank % 4;
    int dr[4] = {-1, 1, 0, 0};
    int dc[4] = {0, 0, -1, 1};
    int move, nr, nc, next, temp, result;

    nodes++;
    if (f > bound) return f;
    if (h == 0) {
        solutionDepth = g;
        return FOUND;
    }

    for (move = 0; move < 4; move++) {
        if (move == (previousMove ^ 1)) continue;
        nr = r + dr[move];
        nc = c + dc[move];
        if (nr < 0 || nr >= 4 || nc < 0 || nc >= 4) continue;

        next = nr * 4 + nc;
        temp = a[blank]; a[blank] = a[next]; a[next] = temp;
        path[g] = move;
        result = dfs(a, next, g + 1, bound, move);
        temp = a[blank]; a[blank] = a[next]; a[next] = temp;

        if (result == FOUND) return FOUND;
        if (result < nextBound) nextBound = result;
    }
    return nextBound;
}

int main() {
    int a[16], i, blank = -1, bound, result;
    char *moves[] = {"Up", "Down", "Left", "Right"};

    printf("Enter the 4x4 puzzle (use 0 for blank):\n");
    for (i = 0; i < 16; i++) {
        scanf("%d", &a[i]);
        if (a[i] == 0) blank = i;
    }

    bound = manhattan(a);
    while (bound < 1000000) {
        result = dfs(a, blank, 0, bound, -1);
        if (result == FOUND) {
            printf("Solution found in %d moves:\n", solutionDepth);
            for (i = 0; i < solutionDepth; i++)
                printf("%s ", moves[path[i]]);
            printf("\nNodes explored: %d\n", nodes);
            return 0;
        }
        if (result == 1000000) break;
        bound = result;
    }

    printf("No solution found (or search limit reached).\n");
    return 0;
}
